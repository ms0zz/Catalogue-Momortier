<body>

  <h1>Mon catalogue</h1>

  <p>Consultez les produits et choisissez les quantités.</p>

  <div id="catalogue"></div>

  <div class="panier">

    <h2>Mon récapitulatif</h2>

    <div id="recap">Aucun article sélectionné.</div>

    <p>Total : <span id="total">0,00 €</span></p>

    <button onclick="afficherRecap()">Voir mon récapitulatif</button>

  </div>

  <script>

    const produits = [

      { nom: "Article 1", prix: 40, quantite: 0 },

      { nom: "Article 2", prix: 20, quantite: 0 },

      { nom: "Article 3", prix: 10, quantite: 0 }

    ];

    const euros = n => n.toLocaleString("fr-FR", {

      style: "currency", currency: "EUR"

    });

    function afficherCatalogue() {

      document.getElementById("catalogue").innerHTML =

        produits.map((p, i) => `

          <div class="produit">

            <h2>${p.nom}</h2>

            <p class="prix">${euros(p.prix)}</p>

            <div class="quantite">

              <button onclick="modifier(${i}, -1)" aria-label="Retirer">−</button>

              <span>${p.quantite}</span>

              <button onclick="modifier(${i}, 1)" aria-label="Ajouter">+</button>

            </div>

          </div>

        `).join("");

      calculerTotal();

    }

    function modifier(i, variation) {

      produits[i].quantite = Math.max(

        0, produits[i].quantite + variation

      );

      afficherCatalogue();

    }

    function calculerTotal() {

      const total = produits.reduce(

        (s, p) => s + p.prix * p.quantite, 0

      );

      document.getElementById("total").textContent = euros(total);

    }

    function afficherRecap() {

      const selection = produits.filter(p => p.quantite > 0);

      document.getElementById("recap").innerHTML = selection.length

        ? selection.map(p =>

            `${p.nom} × ${p.quantite} — ${euros(p.prix * p.quantite)}`

          ).join("<br>")

        : "Aucun article sélectionné.";

    }

    afficherCatalogue();

  </script>
